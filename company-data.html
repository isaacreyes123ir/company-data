const API_BASE = "https://company-data.duckdns.org/api/v1"; let chartEvolucion = null; let chartLiquidez = null; let rucSeleccionado = ""; let datosPrecargados = null; const usd = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 }); const dicClase = { "GEN": "Régimen General", "RMP": "Régimen Impositivo para Microempresas (RIMPE)", "SIM": "Régimen Simplificado Sociedades" }; function corregirCodificacion(texto) { if (!texto) return 'N/A'; try { return decodeURIComponent(escape(texto)); } catch (e) { return texto; } } // LÓGICA DEL BUSCADOR EN VIVO const input = document.getElementById("searchInput"); const list = document.getElementById("autocompleteList"); let timeoutBuscador = null; input.addEventListener("input", function() { const val = this.value.trim(); list.innerHTML = ''; if (!val || val.length < 3) { rucSeleccionado = ""; return; } clearTimeout(timeoutBuscador); timeoutBuscador = setTimeout(async () => { try { const res = await fetch(`${API_BASE}/buscar?q=${encodeURIComponent(val)}`); if (!res.ok) return; const coincidencias = await res.json(); list.innerHTML = ''; coincidencias.forEach(empresa => { const item = document.createElement("div"); item.innerHTML = `${corregirCodificacion(empresa.razon_social)} ${empresa.ruc}`; item.addEventListener("click", function() { input.value = corregirCodificacion(empresa.razon_social); rucSeleccionado = empresa.ruc; list.innerHTML = ''; preCargarEmpresa(rucSeleccionado); }); list.appendChild(item); }); } catch (err) { console.error("Error en búsqueda:", err); } }, 1200); }); document.addEventListener("click", function (e) { if (e.target !== input) list.innerHTML = ''; }); // DESCARGA INVISIBLE EN SEGUNDO PLANO async function preCargarEmpresa(ruc) { datosPrecargados = null; try { const resPerfil = await fetch(`${API_BASE}/perfil/${ruc}`); if (!resPerfil.ok) return; const perfil = await resPerfil.json(); let finanzas = null; const expediente = perfil.directorio ? perfil.directorio.expediente : null; if (expediente) { const resFinanzas = await fetch(`${API_BASE}/finanzas/${expediente}`); if (resFinanzas.ok) { finanzas = await resFinanzas.json(); } } datosPrecargados = { ruc: ruc, perfil: perfil, finanzas: finanzas }; } catch (error) { console.error('Error en precarga:', error); } } // AL PRESIONAR EL BOTÓN "CONSULTAR DATOS" function ejecutarConsultaManual() { const target = rucSeleccionado || input.value.trim(); if(!target) return; if (datosPrecargados && datosPrecargados.ruc === target) { renderizarPerfil(datosPrecargados.perfil); if (datosPrecargados.finanzas) { renderizarFinanzas(datosPrecargados.finanzas); } } else { consultarEmpresa(target); } } async function consultarEmpresa(ruc) { document.getElementById('perfilCard').style.display = 'none'; document.getElementById('finanzasCard').style.display = 'none'; try { const resPerfil = await fetch(`${API_BASE}/perfil/${ruc}`); if (!resPerfil.ok) throw new Error('Empresa no encontrada en registros.'); const perfil = await resPerfil.json(); renderizarPerfil(perfil); const expediente = perfil.directorio ? perfil.directorio.expediente : null; if (expediente) { const resFinanzas = await fetch(`${API_BASE}/finanzas/${expediente}`); if (resFinanzas.ok) { const finanzas = await resFinanzas.json(); renderizarFinanzas(finanzas); } } } catch (error) { alert('Error: ' + error.message); } } function renderizarPerfil(data) { document.getElementById('perfilCard').style.display = 'block'; const sri = data.sri || {}; const dir = data.directorio || {}; document.getElementById('lblRazonSocial').innerText = corregirCodificacion(sri.razon_social || dir.nombre || 'Nombre no disponible'); document.getElementById('lblEstadoSRI').innerText = `SRI: ${sri.estado_contribuyente || 'N/A'}`; document.getElementById('lblEstadoSuper').innerText = `SUPERCIAS: ${dir.situacion_legal || 'N/A'}`; document.getElementById('lblEstadoSRI').style.backgroundColor = sri.estado_contribuyente === 'ACTIVO' ? '#059669' : '#dc2626'; document.getElementById('lblEstadoSuper').style.backgroundColor = (dir.situacion_legal && dir.situacion_legal.includes('ACTIVA')) ? '#2563eb' : '#64748b'; document.getElementById('lblRuc').innerText = sri.numero_ruc || dir.ruc || 'N/A'; document.getElementById('lblExpediente').innerText = dir.expediente || 'N/A'; document.getElementById('lblTipo').innerText = corregirCodificacion(dir.tipo || 'N/A'); const claseRaw = sri.clase_contribuyente ? sri.clase_contribuyente.trim() : ''; document.getElementById('lblClase').innerText = corregirCodificacion(dicClase[claseRaw] || claseRaw || 'N/A'); document.getElementById('lblRepresentante').innerText = corregirCodificacion(dir.representante || 'N/A'); document.getElementById('lblCargo').innerText = corregirCodificacion(dir.cargo || 'N/A'); document.getElementById('lblFechaInicio').innerText = dir.fecha_constitucion || sri.fecha_inicio_actividades || 'N/A'; document.getElementById('lblTelefono').innerText = dir.telefono || 'N/A'; const ubicacion = sri.descripcion_provincia_est ? `${sri.descripcion_provincia_est}, ${sri.descripcion_canton_est}` : (dir.provincia ? `${dir.provincia}, ${dir.canton}` : 'N/A'); const direccion = dir.calle ? `${dir.calle} y ${dir.interseccion || ''}` : 'N/A'; document.getElementById('lblUbicacion').innerText = corregirCodificacion(ubicacion); document.getElementById('lblDireccion').innerText = corregirCodificacion(direccion); document.getElementById('lblObligado').innerText = sri.obligado === 'S' ? 'SÍ' : (sri.obligado === 'N' ? 'NO' : 'N/A'); document.getElementById('lblCapital').innerText = dir.capital_suscrito || '0.00'; document.getElementById('lblCiiu').innerText = dir.ciiu_nivel_6 || 'N/A'; document.getElementById('lblActividad').innerText = corregirCodificacion(sri.actividad_economica || 'N/A'); } function renderizarFinanzas(data) { if(!data.datos || data.datos.length === 0) return; document.getElementById('finanzasCard').style.display = 'block'; const actual = data.datos[0]; document.getElementById('valIngresos').innerText = usd.format(actual.ingresos_ventas); document.getElementById('valUtilidad').innerText = usd.format(actual.utilidad_neta); document.getElementById('valRoe').innerText = `${(actual.roe * 100).toFixed(1)}% / ${(actual.sector_roe * 100).toFixed(1)}%`; document.getElementById('valApalancamiento').innerText = actual.apalancamiento.toFixed(2); document.getElementById('valEmpleados').innerText = actual.n_empleados || 'N/A'; const historico = [...data.datos].reverse(); const anios = historico.map(d => d.anio); if (chartEvolucion) chartEvolucion.destroy(); const ctxEvolucion = document.getElementById('graficoEvolucion').getContext('2d'); chartEvolucion = new Chart(ctxEvolucion, { type: 'line', data: { labels: anios, datasets: [ { label: 'Ingresos por Ventas', data: historico.map(d => d.ingresos_ventas), borderColor: '#2563eb', backgroundColor: '#2563eb', tension: 0.1 }, { label: 'Utilidad Neta', data: historico.map(d => d.utilidad_neta), borderColor: '#dc2626', backgroundColor: '#dc2626', tension: 0.1 } ] }, options: { responsive: true, plugins: { title: { display: true, text: 'Evolución de Ingresos y Utilidad' } } } }); if (chartLiquidez) chartLiquidez.destroy(); const ctxLiquidez = document.getElementById('graficoLiquidez').getContext('2d'); chartLiquidez = new Chart(ctxLiquidez, { type: 'bar', data: { labels: anios, datasets: [ { label: 'Liquidez Corriente', data: historico.map(d => d.liquidez_corriente), backgroundColor: '#10b981' }, { label: 'Prueba Ácida', data: historico.map(d => d.prueba_acida), backgroundColor: '#f59e0b' } ] }, options: { responsive: true, plugins: { title: { display: true, text: 'Liquidez Corriente vs Prueba Ácida' } } } }); }
